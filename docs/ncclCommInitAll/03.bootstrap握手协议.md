# Bootstrap 握手协议

现在讲各个 rank 如何通过 bootstrap network 建立环形拓扑。

## 一、为什么需要 Bootstrap？

**问题**：8 个 ranks 都有了 uniqueId（bootstrap root 的地址），但它们还不知道：
- 其他 ranks 的 IP 地址是什么？
- 哪些 ranks 在同一台机器上？
- 如何建立通信通道？

**Bootstrap 的作用**：
1. **建立初始连接**：所有 ranks 连接到 bootstrap root
2. **交换信息**：每个 rank 告诉 root 自己的地址、GPU 信息等
3. **构建环形拓扑**：Root 协调所有 ranks 形成一个环
4. **AllGather**：通过环形拓扑交换所有 ranks 的完整信息

**核心思想**：用环形拓扑（Ring）进行 AllGather，每个 rank 只需要连接两个邻居（前一个和后一个）。

## 二、核心数据结构

### 1. extInfo（发送给 root 的信息）
```c
struct extInfo {
  int rank;                      // 我的 rank 编号
  int nranks;                    // 总共有多少个 ranks
  int iroot;                     // 我属于哪个 root（支持多 root）
  int nroots;                    // 总共有多少个 roots
  union ncclSocketAddress listenRootAddress;  // 我创建的监听地址（用于接收 root 的回复）
  union ringConnectInfo connectInfo;  // 我的环形连接信息（下一个 rank 连接到我）
};
```

### 2. ringConnectInfo（环形连接信息）
```c
union ringConnectInfo {
  union ncclSocketAddress addr;        // Socket 地址（IP + 端口）
  char handle[NCCL_NET_HANDLE_MAXSIZE];  // ncclNet 句柄（如果使用 InfiniBand）
};
```

### 3. bootstrapState（每个 rank 的状态）
```c
struct bootstrapState {
  struct bootstrapRing_t ring;   // 环形连接（连接到前/后邻居）
  struct bootstrapListen_t listen;  // 监听资源
  int rank;                      // 本 rank 编号
  int nranks;                    // 总 ranks 数
  uint64_t magic;                // 魔数
  union ncclSocketAddress* peerProxyAddresses;  // 所有 ranks 的 proxy 地址
  union ncclSocketAddress* peerP2pAddresses;    // 所有 ranks 的 P2P 地址
  ...
};
```

### 4. bootstrapRootArgs（root 线程参数）
```c
struct bootstrapRootArgs {
  struct ncclSocket* listenSock;  // 用于监听来自 ranks 的连接的 socket
  uint64_t magic;                 // 魔数，用于验证连接的有效性
};
```

## 四、详细步骤（以 3 个 ranks 为例）

### 初始状态
```
uniqueId = { magic: 0xABCD, addr: "192.168.1.100:55555" }

Bootstrap Root 线程:
  - 创建 bootstrapRoot线程，监听在 192.168.1.100:55555
  - 等待连接...

Rank 0, 1, 2:
  - 都拿到了 uniqueId
  - 准备调用 bootstrapInit()
```

---

### Step 1: 每个 rank 创建监听 socket

```c
// Rank 0 执行
bootstrapInit(1, &uniqueId, comm) {
  // 创建监听 socket（用于接收环形邻居的连接）
  createListenSocket(comm, magic, &listenSocket, &myAddr, ncclSocketTypeBootstrap);
  //   └─ socket() → bind(0.0.0.0:0) → listen()
  //   └─ 系统分配端口：假设 60000
  //   └─ myAddr = "192.168.1.100:60000"
  
  // 创建另一个 socket（用于接收 root 的回复）
  createListenSocket(comm, magic, &listenSockRoot, &listenRootAddr, ncclSocketTypeBootstrap);
  //   └─ 系统分配端口：假设 60001
  //   └─ listenRootAddr = "192.168.1.100:60001"
  
  // 填充要发送给 root 的信息
  extInfo info;
  info.rank = 0;
  info.nranks = 3;
  info.nroots = 1;
  info.listenRootAddress = listenRootAddr;  // "192.168.1.100:60001"
  info.connectInfo.addr = myAddr;  // "192.168.1.100:60000"

  ...
  // 后续看 Step 2/4
}
```

同理，Rank 1 和 Rank 2 也创建各自的监听 socket：
```
Rank 0: myAddr = "192.168.1.100:60000", listenRootAddr = "192.168.1.100:60001"
Rank 1: myAddr = "192.168.1.101:60010", listenRootAddr = "192.168.1.101:60011"
Rank 2: myAddr = "192.168.1.102:60020", listenRootAddr = "192.168.1.102:60021"
```

---

### Step 2: 每个 rank 连接到 root 并发送信息

```c
// Rank 0 执行
sendToRoot(BOOTSTRAP_HANDLE(handles, 0), comm, &info) {
  // 连接到 root
  ncclSocketInit(&sock, &handle->addr, handle->magic, ncclSocketTypeBootstrap, ...);
  //   └─ 初始化 socket，目标地址 = "192.168.1.100:55555"
  
  ncclSocketConnect(&sock);
  //   └─ connect(fd, "192.168.1.100:55555", ...)
  //   └─ TCP 三次握手
  //   └─ 验证 magic number
  
  // 发送信息
  socketSend(&sock, &info, sizeof(extInfo));
  //   └─ send(fd, &info, sizeof(extInfo), ...)
  
  ncclSocketClose(&sock);
}
```
Rank 0,1,2 均执行该方法

Root 线程接收：
```c
// 在 bootstrapRoot 线程
do {
  ncclSocketAccept(&sock, listenSock);  // 接受一个连接
  socketRecv(&sock, &info, sizeof(extInfo));  // 接收信息
  
  if (c == 0) {
    // 第一个连接，初始化数据结构
    nranks = info.nranks;  // 3
    nrecv = nranks;  // 期望收到 3 个连接
    ncclCalloc(&rankInfo, nrecv);  // 分配数组存储每个 rank 的信息
    ncclCalloc(&rankAddressesRoot, nrecv);
  }
  
  localId = info.rank;  // 0, 1, 2
  // 保存 rank 的信息
  rankInfo[localId] = info.connectInfo;  // rank 的监听地址
  rankAddressesRoot[localId] = info.listenRootAddress;  // 用于回复的地址
  
  c++;
} while (c < nrecv);  // 直到收到所有 3 个连接
```

收集完成后，Root 线程拿到了全部 Rank 的环形监听地址和 Root 线程回复地址的 IP/Port：
```
rankInfo[0].addr = "192.168.1.100:60000"  (Rank 0 的环形监听地址)
rankInfo[1].addr = "192.168.1.101:60010"  (Rank 1 的环形监听地址)
rankInfo[2].addr = "192.168.1.102:60020"  (Rank 2 的环形监听地址)

rankAddressesRoot[0] = "192.168.1.100:60001"  (Rank 0 等待 root 回复的地址)
rankAddressesRoot[1] = "192.168.1.101:60011"  (Rank 1 等待 root 回复的地址)
rankAddressesRoot[2] = "192.168.1.102:60021"  (Rank 2 等待 root 回复的地址)
```

---

### Step 3: Root 告诉每个 rank 它的"下一个"邻居

Root 构建环形映射：
```
Rank 0 → Rank 1 → Rank 2 → Rank 0 (环)

每个 rank 的"下一个":
  Rank 0 的 next = Rank 1
  Rank 1 的 next = Rank 2
  Rank 2 的 next = Rank 0
```

Root 发送邻居信息：
```c
// 在 bootstrapRoot 线程
for (int r = 0; r < nranks; r++) {
  int next = (r + 1) % nranks;  // 计算下一个邻居
  
  // 发送 rankInfo[next] 给 rankAddressesRoot[r]
  rootSend(&rankAddressesRoot[r], magic, &rankInfo[next]);
  //   └─ 连接到 rankAddressesRoot[r]
  //   └─ 发送 rankInfo[next]
}

// 具体例子：
//   Rank 0: 发送 rankInfo[1] 到 "192.168.1.100:60001"（Rank0 监听的 root 回复地址）
//           → 告诉 Rank 0: 你的下一个邻居是 "192.168.1.101:60010"
//   Rank 1: 发送 rankInfo[2] 到 "192.168.1.101:60011"（Rank1 监听的 root 回复地址）
//           → 告诉 Rank 1: 你的下一个邻居是 "192.168.1.102:60020"
//   Rank 2: 发送 rankInfo[0] 到 "192.168.1.102:60021"（Rank2 监听的 root 回复地址）
//           → 告诉 Rank 2: 你的下一个邻居是 "192.168.1.100:60000"
```

之后 Root 线程的任务结束，退出

---

### Step 4: 每个 rank 接收邻居信息并建立环形连接

```c
// Rank 0 bootstrapInit 中继续执行
ncclSocketAccept(&sock, &listenSockRoot);  // 接受 root 的连接
socketRecv(&sock, &nextPeer, sizeof(nextPeer));  // 接收下一个邻居的信息
//   └─ nextPeer.addr = "192.168.1.101:60010" (Rank 1 的地址)

// 建立环形连接
socketRingConnect(&nextPeer.addr, &sendSocket, &listenSocket, &recvSocket, magic, abortFlag);
//   核心逻辑：
//     1. 连接到下一个邻居（Rank 1）
//        connect(sendSocket, "192.168.1.101:60010")
//
//     2. 接受来自前一个邻居（Rank 2）的连接
//        accept(recvSocket, listenSocket)  // 等待 Rank 2 连接到 "192.168.1.100:60000"
```

所有 ranks 并行执行：
```
Rank 0:
  - sendSocket → 连接到 Rank 1 的 "192.168.1.101:60010"
  - recvSocket ← 等待 Rank 2 连接到 "192.168.1.100:60000"

Rank 1:
  - sendSocket → 连接到 Rank 2 的 "192.168.1.102:60020"
  - recvSocket ← 等待 Rank 0 连接到 "192.168.1.101:60010"

Rank 2:
  - sendSocket → 连接到 Rank 0 的 "192.168.1.100:60000"
  - recvSocket ← 等待 Rank 1 连接到 "192.168.1.102:60020"
```

---

### Step 5: 通过环形拓扑进行 AllGather

现在每个 rank 都有了环形连接，可以交换更多信息（P2P 地址、Proxy 地址等）。

**目标**：让每个 rank 都获得所有其他 ranks 的地址信息。

**算法**：NCCL 使用**双向环形 AllGather**——数据同时向两个方向传播，只需 N/2 步（传统单向需要 N-1 步）。

```c
// Rank 0 执行
// 打包本 rank 的地址信息
ringData[rank].peerAddress = peerP2pAddresses[rank];   // P2P 地址
ringData[rank].peerProxy = peerProxyAddresses[rank];   // Proxy 地址
ringData[rank].peerUDS = peerProxyAddressesUDS[rank];  // UDS 标识符

// 通过环形拓扑进行 AllGather
bootstrapAllGather(state, ringData, sizeof(bootstrapRingData));
//   └─> socketRingAllGather()  // 双向环形算法

// 完成后，ringData 包含所有 ranks 的地址信息
// ringData[0] = Rank 0 的地址
// ringData[1] = Rank 1 的地址
// ringData[2] = Rank 2 的地址
```

**3 个 ranks 的示例**：

```
初始状态:
            slot[0]    slot[1]    slot[2]
Rank 0:    [ A0 ]      [  ]       [  ]
Rank 1:    [  ]        [ A1 ]     [  ]
Rank 2:    [  ]        [  ]       [ A2 ]

Step 0（双向传输）:
  Ring0 (顺时针): Rank 0 发 A0 给 Rank 1，从 Rank 2 收 A2
  Ring1 (逆时针): Rank 0 发 A0 给 Rank 2，从 Rank 1 收 A1

完成后:
            slot[0]    slot[1]    slot[2]
Rank 0:    [ A0 ]      [ A1 ]     [ A2 ]    ✓
Rank 1:    [ A0 ]      [ A1 ]     [ A2 ]    ✓
Rank 2:    [ A0 ]      [ A1 ]     [ A2 ]    ✓
```

**交换的地址信息后续用途**：
- `peerP2pAddresses`：用于 `bootstrapSend/Recv` 点对点通信（建立 GPU 通道时交换连接信息）
- `peerProxyAddresses`：用于 Proxy 线程间通信（处理网络操作，避免阻塞 CUDA kernel）
- `peerProxyAddressesUDS`：节点内优化（Unix Domain Socket 比 TCP 更快）

> 详细的双向环形 AllGather 算法分析见 [03.1.双向环形AllGather.md](03.1.双向环形AllGather.md)

## 五、完整的代码参考

### bootstrapRoot（后台线程）

```c
static void* bootstrapRoot(void* rargs) {
  struct bootstrapRootArgs* args = (struct bootstrapRootArgs*)rargs;
  struct ncclSocket* listenSock = args->listenSock;
  uint64_t magic = args->magic;

  int nranks = 0, c = 0;
  struct extInfo info;
  union ringConnectInfo* rankInfo = NULL;
  union ncclSocketAddress* rankAddressesRoot = NULL;

  // ========== 阶段 1: 收集所有 ranks 的信息 ==========
  do {
    struct ncclSocket sock;
    ncclSocketAccept(&sock, listenSock);  // 接受连接
    socketRecv(&sock, &info, sizeof(info));  // 接收 extInfo
    ncclSocketClose(&sock);

    if (c == 0) {
      // 第一个连接，初始化
      nranks = info.nranks;
      nrecv = nranks + ((nroots > 1) ? 1 : 0);
      ncclCalloc(&rankInfo, nrecv);
      ncclCalloc(&rankAddressesRoot, nrecv);
    }

    localId = localIdFromRoot(info.rank, iroot, nranks, nroots);

    // 保存信息
    rankInfo[localId] = info.connectInfo;
    rankAddressesRoot[localId] = info.listenRootAddress;

    c++;
  } while (c < nrecv);

  // ========== 阶段 2: 发送邻居信息 ==========
  for (int r = 0; r < n2send; r++) {
    int next = (r + 1) % nrecv;
    rootSend(&rankAddressesRoot[r], magic, &rankInfo[next]);
  }

  free(rankInfo);
  free(rankAddressesRoot);
  return NULL;
}
```

### bootstrapInit（每个 rank 执行）

```c
ncclResult_t bootstrapInit(int nHandles, void* handles, struct ncclComm* comm) {
  int rank = comm->rank;
  int nranks = comm->nRanks;
  struct extInfo info = {0};
  union ringConnectInfo nextPeer;

  // ========== Step 1: 创建监听 socket ==========
  createListenSocket(comm, magic, &listenSocket, &myAddr, ncclSocketTypeBootstrap);
  createListenSocket(comm, magic, &listenSockRoot, &listenRootAddr, ncclSocketTypeBootstrap);

  // 填充信息
  info.rank = rank;
  info.nranks = nranks;
  info.listenRootAddress = listenRootAddr;
  info.connectInfo.addr = myAddr;

  // ========== Step 2: 向 root 发送信息 ==========
  sendToRoot(BOOTSTRAP_HANDLE(handles, curr_root), comm, &info);

  // ========== Step 3: 接收 root 的回复 ==========
  ncclSocketAccept(&sock, &listenSockRoot);
  socketRecv(&sock, &nextPeer, sizeof(nextPeer));
  ncclSocketClose(&sock);
  ncclSocketClose(&listenSockRoot);

  // ========== Step 4: 建立环形连接 ==========
  socketRingConnect(&nextPeer.addr, &sendSocket, &listenSocket, &recvSocket, magic, abortFlag);
  //   └─ ncclSocketConnect(sendSocket)  // 连接到下一个 rank
  //   └─ ncclSocketAccept(recvSocket, listenSocket)  // 接受前一个 rank 的连接

  // ========== Step 5: AllGather ==========
  // 创建 Proxy 和 P2P 监听资源
  createListenSocket(comm, magic, proxySocket, peerProxyAddresses + rank, ncclSocketTypeProxy);
  createListenSocket(comm, magic, &peerSocket, peerP2pAddresses + rank, ncclSocketTypeBootstrap);

  // 通过环形拓扑交换所有信息
  ringAllInfo(comm, state, peerP2pAddresses, peerProxyAddresses, peerProxyAddressesUDS, rasRanks);

  return ncclSuccess;
}
```
